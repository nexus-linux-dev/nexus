11macro_rules! old_and_new {
12    ($ty_:expr, $name:ident, $old:ident) => {{
13        $ty_.$old = $ty_.$name;
14        $ty_.$name = $name;
15    }};
16    ($ty_:expr, $name:ident, $old:ident, $path:expr) => {{
17        let _tmp = $path;
18        $ty_.$old = $ty_.$name;
19        $ty_.$name = _tmp;
20    }};
21}

23fn read<P: AsRef<Path>>(parent: P, path: &str, data: &mut [u8]) -> u64 {
24    if let Ok(mut f) = File::open(parent.as_ref().join(path))
25        && let Ok(size) = f.read(data)
26    {
27        let mut i = 0;
28        let mut ret = 0;
29
30        while i < size && i < data.len() && data[i] >= b'0' && data[i] <= b'9' {
31            ret *= 10;
32            ret += (data[i] - b'0') as u64;
33            i += 1;
34        }
35        return ret;
36    }
37    0
38}

40// `read_str` clears and refills the Vec, so its length becomes the length of
41// the string just read. For example, reading "up\n" leaves the Vec length at 3.
42// Keep this buffer separate from numeric read buffers, otherwise counters could
43// be truncated to a few bytes.
44#[allow(clippy::ptr_arg)]
45fn read_str<'data, P: AsRef<Path>>(parent: P, path: &str, data: &'data mut Vec<u8>) -> &'data [u8] {
46    data.clear();
47    if let Ok(mut f) = File::open(parent.as_ref().join(path))
48        && let Ok(size) = f.read_to_end(data)
49    {
50        &mut data[..size]
51    } else {
52        b""
53    }
54}

56impl InterfaceOperationalState {
57    pub(crate) fn from_data(data: &[u8]) -> Self {
58        // see /sys/class/net/<iface>/operstate section at https://www.kernel.org/doc/Documentation/ABI/testing/sysfs-class-net
59        match data {
60            b"unknown" => InterfaceOperationalState::Unknown,
61            b"notpresent" => InterfaceOperationalState::NotPresent,
62            b"down" => InterfaceOperationalState::Down,
63            b"lowerlayerdown" => InterfaceOperationalState::LowerLayerDown,
64            b"testing" => InterfaceOperationalState::Testing,
65            b"dormant" => InterfaceOperationalState::Dormant,
66            b"up" => InterfaceOperationalState::Up,
67            _ => InterfaceOperationalState::Other,
68        }
69    }
70}

72fn refresh_networks_list_from_sysfs(
73    interfaces: &mut HashMap<String, NetworkData>,
74    remove_not_listed_interfaces: bool,
75    sysfs_net: &Path,
76) {
77    if let Ok(dir) = std::fs::read_dir(sysfs_net) {
78        // ilog10 gives number of digits minus one; the extra +1 is for the newline character.
79        let mut num_buf = [0u8; u64::MAX.ilog(10) as usize + 2];
80        let mut str_buf = Vec::with_capacity(32);
81
82        for stats in interfaces.values_mut() {
83            stats.inner.updated = false;
84        }
85
86        for entry in dir.flatten() {
87            let parent = &entry.path().join("statistics");
88            let entry_path = &entry.path();
89            let entry = match entry.file_name().into_string() {
90                Ok(entry) => entry,
91                Err(_) => continue,
92            };
93            let rx_bytes = read(parent, "rx_bytes", &mut num_buf);
94            let tx_bytes = read(parent, "tx_bytes", &mut num_buf);
95            let rx_packets = read(parent, "rx_packets", &mut num_buf);
96            let tx_packets = read(parent, "tx_packets", &mut num_buf);
97            let rx_errors = read(parent, "rx_errors", &mut num_buf);
98            let tx_errors = read(parent, "tx_errors", &mut num_buf);
99            // let rx_compressed = read(parent, "rx_compressed", &mut num_buf);
100            // let tx_compressed = read(parent, "tx_compressed", &mut num_buf);
101            let mtu = read(entry_path, "mtu", &mut num_buf);
102
103            let operational_state = InterfaceOperationalState::from_data(
104                read_str(entry_path, "operstate", &mut str_buf).trim_ascii(),
105            );
106
107            match interfaces.entry(entry) {
108                hash_map::Entry::Occupied(mut e) => {
109                    let interface = e.get_mut();
110                    let interface = &mut interface.inner;
111
112                    old_and_new!(interface, rx_bytes, old_rx_bytes);
113                    old_and_new!(interface, tx_bytes, old_tx_bytes);
114                    old_and_new!(interface, rx_packets, old_rx_packets);
115                    old_and_new!(interface, tx_packets, old_tx_packets);
116                    old_and_new!(interface, rx_errors, old_rx_errors);
117                    old_and_new!(interface, tx_errors, old_tx_errors);
118                    // old_and_new!(e, rx_compressed, old_rx_compressed);
119                    // old_and_new!(e, tx_compressed, old_tx_compressed);
120                    interface.mtu = mtu;
121                    interface.operational_state = operational_state;
122                    interface.updated = true;
123                }
124                hash_map::Entry::Vacant(e) => {
125                    e.insert(NetworkData {
126                        inner: NetworkDataInner {
127                            rx_bytes,
128                            old_rx_bytes: rx_bytes,
129                            tx_bytes,
130                            old_tx_bytes: tx_bytes,
131                            rx_packets,
132                            old_rx_packets: rx_packets,
133                            tx_packets,
134                            old_tx_packets: tx_packets,
135                            rx_errors,
136                            old_rx_errors: rx_errors,
137                            tx_errors,
138                            old_tx_errors: tx_errors,
139                            mac_addr: MacAddr::UNSPECIFIED,
140                            ip_networks: vec![],
141                            // rx_compressed,
142                            // old_rx_compressed: rx_compressed,
143                            // tx_compressed,
144                            // old_tx_compressed: tx_compressed,
145                            operational_state,
146                            mtu,
147                            updated: true,
148                        },
149                    });
150                }
151            };
152        }
153    }
154    // We do this here because `refresh_networks_list_remove_interface` test is checking that
155    // this is working as expected.
156    if remove_not_listed_interfaces {
157        // Remove interfaces which are gone.
158        interfaces.retain(|_, i| {
159            if !i.inner.updated {
160                return false;
161            }
162            i.inner.updated = false;
163            true
164        });
165    }
166}

168pub(crate) struct NetworksInner {
169    pub(crate) interfaces: HashMap<String, NetworkData>,
170}
171
172impl NetworksInner {
173    pub(crate) fn new() -> Self {
174        Self {
175            interfaces: HashMap::new(),
176        }
177    }
178
179    pub(crate) fn list(&self) -> &HashMap<String, NetworkData> {
180        &self.interfaces
181    }
182
183    pub(crate) fn refresh(&mut self, remove_not_listed_interfaces: bool) {
184        refresh_networks_list_from_sysfs(
185            &mut self.interfaces,
186            remove_not_listed_interfaces,
187            Path::new("/sys/class/net/"),
188        );
189        refresh_networks_addresses(&mut self.interfaces);
190    }
191}

193pub(crate) struct NetworkDataInner {
194    /// Total number of bytes received over interface.
195    rx_bytes: u64,
196    old_rx_bytes: u64,
197    /// Total number of bytes transmitted over interface.
198    tx_bytes: u64,
199    old_tx_bytes: u64,
200    /// Total number of packets received.
201    rx_packets: u64,
202    old_rx_packets: u64,
203    /// Total number of packets transmitted.
204    tx_packets: u64,
205    old_tx_packets: u64,
206    /// Shows the total number of packets received with error. This includes
207    /// too-long-frames errors, ring-buffer overflow errors, CRC errors,
208    /// frame alignment errors, fifo overruns, and missed packets.
209    rx_errors: u64,
210    old_rx_errors: u64,
211    /// similar to `rx_errors`
212    tx_errors: u64,
213    old_tx_errors: u64,
214    /// MAC address
215    pub(crate) mac_addr: MacAddr,
216    pub(crate) ip_networks: Vec<IpNetwork>,
217    /// Interface Maximum Transfer Unit (MTU)
218    mtu: u64,
219    operational_state: InterfaceOperationalState,
220    // /// Indicates the number of compressed packets received by this
221    // /// network device. This value might only be relevant for interfaces
222    // /// that support packet compression (e.g: PPP).
223    // rx_compressed: usize,
224    // old_rx_compressed: usize,
225    // /// Indicates the number of transmitted compressed packets. Note
226    // /// this might only be relevant for devices that support
227    // /// compression (e.g: PPP).
228    // tx_compressed: usize,
229    // old_tx_compressed: usize,
230    /// Whether or not the above data has been updated during refresh
231    updated: bool,
232}
233
234impl NetworkDataInner {
235    pub(crate) fn received(&self) -> u64 {
236        self.rx_bytes.saturating_sub(self.old_rx_bytes)
237    }
238
239    pub(crate) fn total_received(&self) -> u64 {
240        self.rx_bytes
241    }
242
243    pub(crate) fn transmitted(&self) -> u64 {
244        self.tx_bytes.saturating_sub(self.old_tx_bytes)
245    }
246
247    pub(crate) fn total_transmitted(&self) -> u64 {
248        self.tx_bytes
249    }
250
251    pub(crate) fn packets_received(&self) -> u64 {
252        self.rx_packets.saturating_sub(self.old_rx_packets)
253    }
254
255    pub(crate) fn total_packets_received(&self) -> u64 {
256        self.rx_packets
257    }
258
259    pub(crate) fn packets_transmitted(&self) -> u64 {
260        self.tx_packets.saturating_sub(self.old_tx_packets)
261    }
262
263    pub(crate) fn total_packets_transmitted(&self) -> u64 {
264        self.tx_packets
265    }
266
267    pub(crate) fn errors_on_received(&self) -> u64 {
268        self.rx_errors.saturating_sub(self.old_rx_errors)
269    }
270
271    pub(crate) fn total_errors_on_received(&self) -> u64 {
272        self.rx_errors
273    }
274
275    pub(crate) fn errors_on_transmitted(&self) -> u64 {
276        self.tx_errors.saturating_sub(self.old_tx_errors)
277    }
278
279    pub(crate) fn total_errors_on_transmitted(&self) -> u64 {
280        self.tx_errors
281    }
282
283    pub(crate) fn mac_address(&self) -> MacAddr {
284        self.mac_addr
285    }
286
287    pub(crate) fn ip_networks(&self) -> &[IpNetwork] {
288        &self.ip_networks
289    }
290
291    pub(crate) fn mtu(&self) -> u64 {
292        self.mtu
293    }
294
295    pub(crate) fn operational_state(&self) -> InterfaceOperationalState {
296        self.operational_state
297    }
298}

306    #[test]
307    fn refresh_networks_list_add_interface() {
308        let sys_net_dir = tempfile::tempdir().expect("failed to create temporary directory");
309
310        fs::create_dir(sys_net_dir.path().join("itf1")).expect("failed to create subdirectory");
311
312        let mut interfaces = HashMap::new();
313
314        refresh_networks_list_from_sysfs(&mut interfaces, false, sys_net_dir.path());
315        assert_eq!(interfaces.keys().collect::<Vec<_>>(), ["itf1"]);
316
317        fs::create_dir(sys_net_dir.path().join("itf2")).expect("failed to create subdirectory");
318
319        refresh_networks_list_from_sysfs(&mut interfaces, false, sys_net_dir.path());
320        let mut itf_names: Vec<String> = interfaces.keys().map(|n| n.to_owned()).collect();
321        itf_names.sort();
322        assert_eq!(itf_names, ["itf1", "itf2"]);
323    }
324
325    #[test]
326    fn refresh_networks_list_remove_interface() {
327        let sys_net_dir = tempfile::tempdir().expect("failed to create temporary directory");
328
329        let itf1_dir = sys_net_dir.path().join("itf1");
330        let itf2_dir = sys_net_dir.path().join("itf2");
331        fs::create_dir(&itf1_dir).expect("failed to create subdirectory");
332        fs::create_dir(itf2_dir).expect("failed to create subdirectory");
333
334        let mut interfaces = HashMap::new();
335
336        refresh_networks_list_from_sysfs(&mut interfaces, false, sys_net_dir.path());
337        let mut itf_names: Vec<String> = interfaces.keys().map(|n| n.to_owned()).collect();
338        itf_names.sort();
339        assert_eq!(itf_names, ["itf1", "itf2"]);
340
341        fs::remove_dir(&itf1_dir).expect("failed to remove subdirectory");
342
343        refresh_networks_list_from_sysfs(&mut interfaces, true, sys_net_dir.path());
344        assert_eq!(interfaces.keys().collect::<Vec<_>>(), ["itf2"]);
345    }

347    #[test]
348    fn refresh_networks_list_with_multiple_interfaces() {
349        let dir = tempfile::tempdir().expect("failed to create temporary directory");
350
351        for (name, rx) in [
352            ("if_a", "100"),
353            ("if_b", "1234567890123"),
354            ("if_c", "9876543210987"),
355            ("if_d", "18446744073709551615"),
356        ] {
357            let stats = dir.path().join(name).join("statistics");
358
359            fs::create_dir_all(&stats).expect("failed to create statistics dir");
360
361            for f in &[
362                "rx_bytes",
363                "tx_bytes",
364                "rx_packets",
365                "tx_packets",
366                "rx_errors",
367                "tx_errors",
368            ] {
369                fs::write(stats.join(f), rx).expect("failed to write stats");
370            }
371
372            fs::write(dir.path().join(name).join("mtu"), "1500").expect("failed to write mtu");
373
374            fs::write(dir.path().join(name).join("operstate"), "up\n")
375                .expect("failed to write operstate");
376        }
377
378        let mut interfaces = HashMap::new();
379
380        refresh_networks_list_from_sysfs(&mut interfaces, false, dir.path());
381        refresh_networks_list_from_sysfs(&mut interfaces, false, dir.path());
382
383        assert_eq!(interfaces.get("if_a").unwrap().inner.rx_bytes, 100);
384        assert_eq!(
385            interfaces.get("if_b").unwrap().inner.rx_bytes,
386            1_234_567_890_123
387        );
388        assert_eq!(
389            interfaces.get("if_c").unwrap().inner.rx_bytes,
390            9_876_543_210_987
391        );
392        assert_eq!(interfaces.get("if_d").unwrap().inner.rx_bytes, u64::MAX);
393
394        for name in ["if_a", "if_b", "if_c", "if_d"] {
395            let interface = interfaces.get(name).unwrap();
396            assert_eq!(interface.inner.mtu, 1500, "{name}: mtu");
397        }
398    }
