9pub(crate) struct UserInner {
10    pub(crate) uid: Uid,
11    pub(crate) gid: Gid,
12    pub(crate) name: String,
13    c_user: Vec<u8>,
14}

16impl UserInner {
17    pub(crate) fn new(uid: Uid, gid: Gid, name: String) -> Self {
18        let mut c_user = name.as_bytes().to_vec();
19        c_user.push(0);
20        Self {
21            uid,
22            gid,
23            name,
24            c_user,
25        }
26    }
27
28    pub(crate) fn id(&self) -> &Uid {
29        &self.uid
30    }
31
32    pub(crate) fn group_id(&self) -> Gid {
33        self.gid
34    }
35
36    pub(crate) fn name(&self) -> &str {
37        &self.name
38    }
39
40    pub(crate) fn groups(&self) -> Vec<Group> {
41        unsafe { get_user_groups(self.c_user.as_ptr() as *const _, self.gid.0 as _) }
42    }
43}

45pub(crate) unsafe fn get_group_name(
46    id: libc::gid_t,
47    buffer: &mut Vec<libc::c_char>,
48) -> Option<String> {
49    let mut g = std::mem::MaybeUninit::<libc::group>::uninit();
50    let mut tmp_ptr = std::ptr::null_mut();
51    let mut last_errno = 0;
52
53    unsafe {
54        loop {
55            if retry_eintr!(set_to_0 => last_errno => getgrgid_r(
56                id as _,
57                g.as_mut_ptr() as _,
58                buffer.as_mut_ptr(),
59                buffer.capacity() as _,
60                &mut tmp_ptr as _
61            )) != 0
62            {
63                // If there was not enough memory, we give it more.
64                if last_errno == libc::ERANGE as libc::c_int {
65                    // Needs to be updated for `Vec::reserve` to actually add additional capacity.
66                    // In here it's "fine" since we never read from `buffer`.
67                    buffer.set_len(buffer.capacity());
68                    buffer.reserve(2048);
69                    continue;
70                }
71                return None;
72            }
73            break;
74        }
75        let g = g.assume_init();
76        super::utils::cstr_to_rust(g.gr_name)
77    }
78}

80pub(crate) unsafe fn get_user_groups(
81    name: *const libc::c_char,
82    group_id: libc::gid_t,
83) -> Vec<Group> {
84    let mut buffer = Vec::with_capacity(2048);
85    let mut groups = Vec::with_capacity(256);
86
87    let mut nb_groups = groups.capacity() as c_int;
88    let mut old_nb_groups = nb_groups;
89    loop {
90        unsafe {
91            if getgrouplist(name, group_id as _, groups.as_mut_ptr(), &mut nb_groups) == -1 {
92                // prevent infinite looping.
93                if old_nb_groups >= nb_groups {
94                    sysinfo_debug!(
95                        "getgrouplist failed but buffer size requested was not larger than existing buffer"
96                    );
97                    return Vec::new();
98                }
99                groups.reserve(nb_groups as usize);
100                // reserve could reserve more capacity than requestes, so this could
101                // overflow c_int.
102                nb_groups = groups.capacity().try_into().unwrap_or(c_int::MAX);
103                old_nb_groups = nb_groups;
104                continue;
105            }
106            groups.set_len(nb_groups as _);
107            return groups
108                .iter()
109                .filter_map(|group_id| {
110                    let name = get_group_name(*group_id as _, &mut buffer)?;
111                    Some(Group {
112                        inner: crate::GroupInner::new(Gid(*group_id as _), name),
113                    })
114                })
115                .collect();
116        }
117    }
118}

120#[cfg(not(any(target_os = "macos", target_os = "ios")))]
121pub(crate) fn get_users(users: &mut Vec<crate::User>) {
122    use std::io::{BufRead, BufReader};
123    use std::str::FromStr;
124
125    fn filter(shell: &str, uid: libc::uid_t) -> bool {
126        uid < 65536 && !shell.ends_with("/false") && !shell.ends_with("/uucico")
127    }
128
129    users.clear();
130
131    // We cannot use `getpwent`, `setpwent` and `endpwent` because they're not thread-safe. The
132    // `getpwent_r` equivalent is not thread-safe either. So we have to retrieve the users list
133    // ourselves...
134    let Ok(file) = std::fs::File::open("/etc/passwd") else {
135        sysinfo_debug!("failed to open `/etc/passwd`");
136        return;
137    };
138    let mut users_map = std::collections::HashMap::with_capacity(10);
139
140    for line in BufReader::new(file).lines() {
141        let Ok(line) = line else { continue };
142        let mut parts = line.split(':');
143        let Some(name) = parts.next() else { continue };
144        if users_map.contains_key(name) {
145            continue;
146        }
147        parts.next(); // We skip the password field.
148        let Some(uid) = parts.next().and_then(|v| libc::uid_t::from_str(v).ok()) else {
149            continue;
150        };
151        let Some(gid) = parts.next().and_then(|v| libc::gid_t::from_str(v).ok()) else {
152            continue;
153        };
154        parts.next(); // We skip the "comment" field.
155        parts.next(); // We skip the directory field.
156        let Some(shell) = parts.next() else { continue };
157
158        if !filter(shell, uid) {
159            continue;
160        }
161        users_map.insert(name.into(), (Uid(uid), Gid(gid)));
162    }
163
164    for (name, (uid, gid)) in users_map {
165        users.push(crate::User {
166            inner: UserInner::new(uid, gid, name),
167        });
168    }
169}
